# 后端基础对象（建议最小集合）

> 目标：提供后端“基础对象文件”，用于统一地图模块的核心业务对象定义。
> 范围：Route / Point / Fence + 通用坐标类型（不含渲染聚合 DTO）。

## 基础业务对象
- Route（路线）
- Point（点位）
- Fence（电子围栏）

## 通用类型
- LngLat（经纬度坐标，lng=经度，lat=纬度）
- Bound（外接矩形范围，sw=西南角，ne=东北角）
- Polyline（路线几何点 / 轨迹点 / 贴路网点）
- ISODateTime（日期时间字符串）
- ISODate（日期字符串）

## 系统基础对象结构

### 通用类型
```ts
// ISO 8601 日期时间，例如：2026-02-09T10:00:00Z
export type ISODateTime = string;

// ISO 日期，YYYY-MM-DD
export type ISODate = string;

// 经纬度坐标
export interface LngLat {
  lng: number; // 经度
  lat: number; // 纬度
}

// 外接矩形范围（地图 fitBounds）
export interface Bound {
  sw: LngLat; // 西南角
  ne: LngLat; // 东北角
}

// 贴路网点（也可叫做轨迹点或路径点）
export type Polyline = LngLat[] | string;
```

### Fence（电子围栏）
```ts
// 围栏类型
export enum FenceType {
  CIRCLE = "CIRCLE",   // 圆形围栏
  POLYGON = "POLYGON", // 多边形围栏
}

// 围栏状态
export enum FenceStatus {
  ENABLED = "ENABLED",   // 启用
  DISABLED = "DISABLED", // 禁用
}

// 围栏对象
export interface FenceDTO {
  id: string;          // 围栏ID
  name: string;        // 围栏名称
  type: FenceType;     // 围栏类型
  status: FenceStatus; // 启用状态

  center?: LngLat | null;   // 圆心（仅 type=CIRCLE）
  radius?: number | null;   // 半径，米（仅 type=CIRCLE）
  points?: LngLat[] | null; // 多边形点集（仅 type=POLYGON）

  bound?: Bound | null;     // 外接矩形（用于 fitBounds）
  remark?: string | null;   // 备注
  createdAt?: ISODateTime;  // 创建时间
  updatedAt?: ISODateTime;  // 更新时间
}
```

### Point（点位）
```ts
// 打卡点位对象
export interface PointDTO {
  id: string;                 // 打卡点位ID
  name: string;               // 打卡点位名称
  location: LngLat;           // 打卡点位坐标
  address?: string | null;    // 逆地理编码地址
  fenceIds?: string[] | null; // 关联围栏ID列表
  remark?: string | null;     // 备注
  createdAt?: ISODateTime;    // 创建时间
  updatedAt?: ISODateTime;    // 更新时间
}
```

### Route（路线）
```ts
// 出行方式
export enum TravelMode {
  BICYCLING = "BICYCLING", // 骑行
}

// 路线点位引用
export interface RoutePointRef {
  pointId: string;     // 点位ID
  order: number;       // 顺序号（从 1 开始）
  name?: string;       // 点位名称（可选，用于展示）
  location?: LngLat;   // 点位坐标（可选，用于打点）
  fenceIds?: string[]; // 关联围栏ID（可选，用于沿线渲染）
}

// 路线对象
export interface RouteDTO {
  id: string;               // 路线ID
  name: string;             // 路线名称
  travelMode: TravelMode;   // 出行方式
  points: RoutePointRef[];  // 点位顺序集合
  remark?: string | null;   // 备注
  createdAt?: ISODateTime;  // 创建时间
  updatedAt?: ISODateTime;  // 更新时间
}
```





## 不属于基础对象的部分

以下内容为接口层/聚合层 DTO 或业务扩展模块

### 推荐排序 / 预览计算
```ts
// 推荐排序请求
export interface RecommendOrderRequest {
  travelMode: TravelMode;            // 出行方式
  pointIds: string[];                // 参与排序的点位ID集合（无序）
  fixedStartPointId?: string | null; // 固定起点（可选）
  fixedEndPointId?: string | null;   // 固定终点（可选）
  needPolyline?: boolean;            // 是否需要返回路线几何（可选）
}

// 推荐排序分段
export interface RecommendOrderSegment {
  fromPointId: string;          // 分段起点ID
  toPointId: string;            // 分段终点ID
  theoryDistanceMeter: number;  // 理论距离（米）
  theoryDurationSecond: number; // 理论耗时（秒）
  polyline?: Polyline;          // 分段几何（可选）
}

// 推荐排序响应
export interface RecommendOrderResponse {
  orderedPointIds: string[];          // 排序后的点位ID
  segments: RecommendOrderSegment[];  // 分段详情
  totalTheoryDistanceMeter: number;   // 总理论距离（米）
  totalTheoryDurationSecond: number;  // 总理论耗时（秒）
  bound?: Bound;                      // 外接矩形（可选）
  polyline?: Polyline;                // 总体路线几何（可选）
}
```

### 路线详情 / 预览
```ts
// 路线预览（用于详情页地图展示）
export interface RoutePreviewDTO {
  bound?: Bound;                      // 外接矩形（建议返回）
  polyline?: Polyline;                // 贴路网路线几何（可选）
  segments?: RecommendOrderSegment[]; // 分段详情（可选）
  totalTheoryDistanceMeter?: number;  // 总理论距离（可选）
  totalTheoryDurationSecond?: number; // 总理论耗时（可选）
}

// 路线详情返回（可包含预览）
export interface RouteDetailDTO extends RouteDTO {
  preview?: RoutePreviewDTO; // needPolyline=true 时返回
}
```

### 路线渲染聚合（单次渲染）
```ts
// 路线渲染聚合 DTO（路线 + 围栏 + 贴路网 polyline + 点位）
export interface RouteRenderDTO {
  route: {
    id: string;                     // 路线ID
    name: string;                   // 路线名称
    travelMode: TravelMode;         // 出行方式
    bound?: Bound;                  // 外接矩形（可选）
    polyline: LngLat[];             // 贴路网路线几何（点集）
    points: Array<{
      id: string;                   // 点位ID
      name: string;                 // 点位名称
      location: LngLat;             // 点位坐标
      fenceIds?: string[];          // 关联围栏ID（可选）
    }>;
    segments?: RecommendOrderSegment[]; // 分段详情（可选）
    totalTheoryDistanceMeter?: number;  // 总理论距离（可选）
    totalTheoryDurationSecond?: number; // 总理论耗时（可选）
  };
  fences: FenceDTO[];               // 路线相关围栏集合
}
```

### 导出任务模块

// 导出状态
export enum ExportStatus {
  PENDING = "PENDING", // 待处理

// 导出文件格式

export enum ExportType {

export interface ExportCreateRequest {

// 创建导出任务响应
export interface ExportCreateResponse {
  taskId: string; // 任务ID
}

// 导出任务对象
export interface ExportTaskDTO {
  taskId: string;                 // 任务ID

  downloadUrl?: string | null;    // 下载地址（可选）
